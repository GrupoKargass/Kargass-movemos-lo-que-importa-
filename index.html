// authService.js - Gestión de sesión, persistencia de ruta y control de inactividad import AsyncStorage from '@async-storage/async-storage'; const SESSION_TIMEOUT_HOURS = 8; // Tiempo máximo de inactividad antes de pedir contraseña (ej: 8 horas) const STORAGE_KEYS = { TOKEN: '@kargass_auth_token', USER_PLATE: '@kargass_user_plate', LAST_ACTIVITY: '@kargass_last_activity', LAST_ROUTE: '@kargass_last_route', EXPLICIT_LOGOUT: '@kargass_explicit_logout', }; export const authService = { // 1. Guardar sesión exitosa al iniciar sesión async loginSuccess(token, plate, initialRoute = '/driver/dashboard') { const now = new Date().getTime(); await AsyncStorage.multiSet([ [STORAGE_KEYS.TOKEN, token], [STORAGE_KEYS.USER_PLATE, plate], [STORAGE_KEYS.LAST_ACTIVITY, now.toString()], [STORAGE_KEYS.LAST_ROUTE, initialRoute], [STORAGE_KEYS.EXPLICIT_LOGOUT, 'false'], ]); }, // 2. Actualizar la última actividad y ruta actual (cada vez que navega o usa la app) async updateActivity(currentRoute) { const now = new Date().getTime(); await AsyncStorage.setItem(STORAGE_KEYS.LAST_ACTIVITY, now.toString()); if (currentRoute) { await AsyncStorage.setItem(STORAGE_KEYS.LAST_ROUTE, currentRoute); } }, // 3. Verificar estado de la sesión al abrir la aplicación async checkSessionState() { try { const token = await AsyncStorage.getItem(STORAGE_KEYS.TOKEN); const explicitLogout = await AsyncStorage.getItem(STORAGE_KEYS.EXPLICIT_LOGOUT); const lastActivity = await AsyncStorage.getItem(STORAGE_KEYS.LAST_ACTIVITY); const savedPlate = await AsyncStorage.getItem(STORAGE_KEYS.USER_PLATE) || ''; const lastRoute = await AsyncStorage.getItem(STORAGE_KEYS.LAST_ROUTE) || '/driver/dashboard'; // Si cerró sesión manualmente o no hay token, requerir login completo if (explicitLogout === 'true' || !token || !lastActivity) { return { status: 'REQUIRE_FULL_LOGIN', plate: savedPlate }; } // Validar tiempo de inactividad (Timeout) const now = new Date().getTime(); const hoursElapsed = (now - parseInt(lastActivity, 10)) / (1000 * 60 * 60); if (hoursElapsed > SESSION_TIMEOUT_HOURS) { // Expiró por tiempo: Limpiamos token pero conservamos la placa await AsyncStorage.removeItem(STORAGE_KEYS.TOKEN); return { status: 'REQUIRE_PASSWORD_ONLY', plate: savedPlate }; } // Sesión viva: Retornamos la ruta donde iba el conductor return { status: 'RESTORE_SESSION', plate: savedPlate, route: lastRoute }; } catch (error) { console.error('Error verificando la sesión:', error); return { status: 'REQUIRE_FULL_LOGIN', plate: '' }; } }, // 4. Salida manual explícita (Cerrar sesión) async logoutManually() { const savedPlate = await AsyncStorage.getItem(STORAGE_KEYS.USER_PLATE) || ''; await AsyncStorage.multiRemove([ STORAGE_KEYS.TOKEN, STORAGE_KEYS.LAST_ACTIVITY, STORAGE_KEYS.LAST_ROUTE, ]); await AsyncStorage.setItem(STORAGE_KEYS.EXPLICIT_LOGOUT, 'true'); return savedPlate; // Retorna la placa para dejarla autofill en el login } };
